Step 1.1 — Create steps/api/clients/list.step.ts
GET /api/clients — Returns all clients with their industry and primary contact
1
What this file does
When the frontend calls GET /api/clients, this step handler runs. It authenticates the request (verifies the JWT), then queries PostgreSQL through Prisma to get all clients with their related industry and contacts. It returns the list as JSON.
2
The pattern — copy from your existing deals/list.step.ts
Your deal list step is the exact template. The only differences are: the Prisma model (prisma.client instead of prisma.deal), the include fields, and the URL path.
steps/api/clients/list.step.ts
import { StepConfig, Handlers } from 'motia'
import { authenticate } from '../../../lib/auth'
import { prisma } from '../../../lib/db'

export const config = {
  name: 'ListClients',
  description: 'Get list of all clients',
  triggers: [
    { type: 'http', method: 'GET', path: '/api/clients' },
  ],
  enqueues: [],
  flows: ['sales-pipeline'],
} as const satisfies StepConfig

export const handler: Handlers<typeof config> = async (req, { logger }) => {
  const user = await authenticate(req)
  logger.info('Listing clients', { userId: user.id })

  const clients = await prisma.client.findMany({
    include: {
      industry: true,                         // join the industry name
      contacts: true,                         // all contacts for this client
      contact: true,                          // the primary contact
      _count: { select: { deals: true } },   // how many deals
    },
    orderBy: { name: 'asc' },
  })

  return { status: 200, body: clients }
}
What's happening line by line:
• config.triggers — tells Motia "when someone hits GET /api/clients, run this handler"
• authenticate(req) — extracts JWT from header, verifies it, returns the logged-in BD user
• prisma.client.findMany — generates SQL: SELECT * FROM client LEFT JOIN industry...
• include — tells Prisma to also fetch related tables (like SQL JOINs)
• _count: { select: { deals: true } } — adds a _count.deals field so the UI can show "5 deals"

import { StepConfig, Handlers } from 'motia'
import { authenticate } from '../../../lib/auth'
import { prisma } from '../../../lib/db'

export const config = {
  name: 'GetClientDetail',
  description: 'Get a single client by ID',
  triggers: [
    { type: 'http', method: 'GET', path: '/api/clients/:id' },
  ],
  enqueues: [],
  flows: ['sales-pipeline'],
} as const satisfies StepConfig

export const handler: Handlers<typeof config> = async (req, { logger }) => {
  const user = await authenticate(req)
  const { id } = req.pathParams      // Motia extracts :id from the URL

  const client = await prisma.client.findUnique({
    where: { id },
    include: {
      industry: true,
      contact: true,                          // primary contact
      contacts: true,                         // all contacts
      deals: {                                  // all deals for this client
        include: {
          stage: true,
          bd: { select: { id: true, firstName: true, lastName: true } },
        },
      },
      referredBy: true,                      // who referred this client
    },
  })

  if (!client) {
    return { status: 404, body: { error: 'Client not found' } }
  }

  return { status: 200, body: client }
}
Key difference from list: findUnique returns one record (or null). We check for null and return 404. The include is richer — on a detail page, you want everything (deals, all contacts, referral info). On the list page, you want less to keep the payload small.
Step 1.3 — Create steps/api/clients/create.step.ts
POST /api/clients — Creates a new client company
1
What this file does
When a user fills out the ClientForm and clicks submit, the frontend POSTs to this endpoint. Zod validates the body (so bad data gets rejected with clear error messages). Then Prisma inserts the row.
steps/api/clients/create.step.ts
import { StepConfig, Handlers } from 'motia'
import { z } from 'zod'
import { authenticate } from '../../../lib/auth'
import { prisma } from '../../../lib/db'

export const config = {
  name: 'CreateClient',
  description: 'Create a new client',
  triggers: [{
    type: 'http',
    method: 'POST',
    path: '/api/clients',
    bodySchema: z.object({                     // Zod validates BEFORE handler runs
      name:        z.string().min(1),           // required
      brand:       z.string().optional(),
      accountType: z.enum(['ENTERPRISE', 'CORPORATE', 'SMB', 'GOVERNMENT']),
      status:      z.enum(['ACTIVE', 'INACTIVE', 'PROSPECT']).default('PROSPECT'),
      industryId:  z.string().optional(),
      referralId:  z.string().optional(),
    }),
  }],
  enqueues: [],
  flows: ['sales-pipeline'],
} as const satisfies StepConfig

export const handler: Handlers<typeof config> = async (req, { logger }) => {
  const user = await authenticate(req)
  const { name, brand, accountType, status, industryId, referralId } = req.body

  const client = await prisma.client.create({
    data: { name, brand, accountType, status, industryId, referralId },
    include: { industry: true, contacts: true },
  })

  logger.info('Client created', { clientId: client.id, by: user.id })
  return { status: 201, body: client }
}
About Zod validation: The bodySchema in the trigger config means Motia automatically validates the request body before your handler runs. If someone sends { name: "" }, they get a 400 error with details like "name must be at least 1 character". You never have to write manual validation in the handler.
Step 1.4 — Create steps/api/clients/update.step.ts
PATCH /api/clients/:id — Update an existing client
1
What this file does
PATCH means "partial update" — the frontend only sends the fields that changed. If the user only changes the status, the body is just { status: "ACTIVE" }. All fields are optional in the Zod schema.
steps/api/clients/update.step.ts
import { StepConfig, Handlers } from 'motia'
import { z } from 'zod'
import { authenticate } from '../../../lib/auth'
import { prisma } from '../../../lib/db'

export const config = {
  name: 'UpdateClient',
  description: 'Update an existing client',
  triggers: [{
    type: 'http',
    method: 'PATCH',
    path: '/api/clients/:id',
    bodySchema: z.object({
      name:        z.string().min(1).optional(),   // all optional for partial update
      brand:       z.string().optional(),
      accountType: z.enum(['ENTERPRISE', 'CORPORATE', 'SMB', 'GOVERNMENT']).optional(),
      status:      z.enum(['ACTIVE', 'INACTIVE', 'PROSPECT']).optional(),
      industryId:  z.string().optional(),
      contactId:   z.string().optional(),          // set primary contact
    }),
  }],
  enqueues: [],
  flows: ['sales-pipeline'],
} as const satisfies StepConfig

export const handler: Handlers<typeof config> = async (req, { logger }) => {
  const user = await authenticate(req)
  const { id } = req.pathParams

  const existing = await prisma.client.findUnique({ where: { id } })
  if (!existing) {
    return { status: 404, body: { error: 'Client not found' } }
  }

  const updated = await prisma.client.update({
    where: { id },
    data: req.body,                           // Zod already validated — safe to spread
    include: { industry: true, contacts: true, contact: true },
  })

  logger.info('Client updated', { clientId: id, by: user.id })
  return { status: 200, body: updated }
}

frontend/src/hooks/useClients.ts — Rewrite
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import { clientsApi } from '../api/clients'

export function useClients() {
  return useQuery({
    queryKey: ['clients'],
    queryFn: async () => {
      const res = await clientsApi.list()
      return res.data
    },
  })
}

export function useClient(id: string) {
  return useQuery({
    queryKey: ['client', id],
    queryFn: async () => {
      const res = await clientsApi.getById(id)
      return res.data
    },
    enabled: !!id,          // don't fetch if no ID yet
  })
}

export function useCreateClient() {
  const qc = useQueryClient()
  return useMutation({
    mutationFn: async (data: any) => {
      const res = await clientsApi.create(data)
      return res.data
    },
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ['clients'] })  // refresh the list
    },
  })
}

export function useUpdateClient() {
  const qc = useQueryClient()
  return useMutation({
    mutationFn: async ({ id, data }: { id: string; data: any }) => {
      const res = await clientsApi.update(id, data)
      return res.data
    },
    onSuccess: (_, { id }) => {
      qc.invalidateQueries({ queryKey: ['clients'] })
      qc.invalidateQueries({ queryKey: ['client', id] })
    },
  })
}
What changed: Instead of useState(MOCK_CLIENTS), we now use useQuery which calls your real API. The magic of React Query: after you create or update a client, invalidateQueries tells React Query "the clients list is stale" → it automatically refetches → the UI updates. You don't have to manually manage state.

Update pages to use the new hooks

Step 3.3 — steps/api/deals/stage.step.ts ⭐ Most Important
PATCH /api/deals/:id/stage — Move a deal through the pipeline + audit log
1
Why this is special
A stage change isn't just an update — it needs to: close the previous audit log entry, create a new one, update the deal's stage, and handle special cases (Closed Won/Lost sets isClosed and closedDate). This is a transaction — all of it succeeds or none of it does.
steps/api/deals/stage.step.ts
import { StepConfig, Handlers } from 'motia'
import { z } from 'zod'
import { authenticate } from '../../../lib/auth'
import { prisma } from '../../../lib/db'

export const config = {
  name: 'UpdateDealStage',
  description: 'Move a deal to a new pipeline stage',
  triggers: [{
    type: 'http',
    method: 'PATCH',
    path: '/api/deals/:id/stage',
    bodySchema: z.object({
      stageId: z.string().min(1),
      notes:   z.string().optional(),   // reason for moving
    }),
  }],
  enqueues: [],
  flows: ['sales-pipeline'],
} as const satisfies StepConfig

export const handler: Handlers<typeof config> = async (req, { logger }) => {
  const user = await authenticate(req)
  const { id } = req.pathParams
  const { stageId, notes } = req.body

  // Use a transaction — all or nothing
  const result = await prisma.$transaction(async (tx) => {
    const deal = await tx.deal.findUnique({
      where: { id },
      include: { stage: true },
    })
    if (!deal) throw new Error('Deal not found')

    const newStage = await tx.pipelineStage.findUnique({
      where: { id: stageId },
    })
    if (!newStage) throw new Error('Stage not found')

    const now = new Date()

    // 1. Close the current audit log entry
    const currentLog = await tx.dealAuditLog.findFirst({
      where: { dealId: id, exitedAt: null },
    })
    if (currentLog) {
      const days = Math.floor(
        (now.getTime() - currentLog.enteredAt.getTime()) / (86400000)
      )
      await tx.dealAuditLog.update({
        where: { id: currentLog.id },
        data: { exitedAt: now, daysInStage: days },
      })
    }

    // 2. Create new audit log entry
    await tx.dealAuditLog.create({
      data: {
        dealId: id, stageId, enteredAt: now,
        changedById: user.id, notes,
      },
    })

    // 3. Update the deal itself
    const isClosed = ['Closed Won', 'Closed Lost'].includes(newStage.name)
    const updated = await tx.deal.update({
      where: { id },
      data: {
        stageId,
        lastStageUpdateAt: now,
        ...(isClosed && { isClosed: true, closedDate: now }),
      },
      include: {
        client: true, stage: true,
        bd: { select: { id: true, firstName: true, lastName: true }},
      },
    })

    return updated
  })

  return { status: 200, body: result }
}
