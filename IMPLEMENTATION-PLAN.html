1. Assumptions & Strategic Decision

Key Assumptions

  • Zeandy: 4–5 hrs/day, 7 days = ~32 hrs total
  • Andre: starting from scratch in Python, similar hours = ~32 hrs total
  • PostgreSQL is running locally with seeded data (pipeline stages, industries, services, 4 BD accounts)
  • No test framework — we skip automated tests this sprint and rely on curl + manual UI smoke tests
  • No CI/CD — manual deploy; Railway push on Day 6–7 if time allows
  • Conventional Commits: feat:, fix:, chore:
  • Demo audience: stakeholders / leadership who want to see the working system end-to-end

Critical Strategy Decision: Two-Phase Integration

Andre hasn't started yet and needs to scaffold a Python/FastAPI service from zero. Building a separate service + JWT verification + CORS + deployment in 7 days introduces high integration risk for the demo.

Phase 1 (This Week — Demo-Safe): Zeandy implements dashboard/report endpoints as Motia TypeScript steps using Prisma raw queries. This keeps the demo on a single service with zero integration risk. Meanwhile, Andre builds the proper Python analytics service in parallel.

Phase 2 (Post-Demo): Once Andre's FastAPI service is tested, swap the Motia dashboard steps for Andre's Python API. The frontend doesn't change — only the backend endpoint origin swaps.

Why this works: The JSON response shapes are agreed upon now. Zeandy's Motia steps return the exact same shape Andre's Python API will return. The frontend is identical either way.

5.2.1 Deal updateStage Step (Day 1)

// steps/api/deals/updateStage.step.ts
import { type Handlers, type StepConfig, logger } from 'motia'
import { z } from 'zod'
import prisma from '../../../lib/db'
import { authenticate, AuthError } from '../../../lib/auth'

const STAGE_PROBABILITY: Record<string, number> = {
  'Inquiry': 10, 'Prospecting': 20, 'Discovery': 40,
  'Proposal Sent': 60, 'Negotiation': 75,
  'Closed Won': 100, 'Closed Lost': 0,
}

export const config: StepConfig = {
  type: 'api',
  name: 'Update Deal Stage',
  description: 'Move a deal to a new pipeline stage with audit logging',
  triggers: [{ type: 'api', method: 'PATCH', path: '/api/deals/:id/stage' }],
  bodySchema: z.object({
    stageId: z.string().uuid(),
    notes: z.string().optional(),
  }),
}

export const handler: Handlers['api'] = async (req, { emit }) => {
  try {
    const user = await authenticate(req.request)
    const { id } = req.pathParams
    const { stageId, notes } = req.body

    // Validate deal exists and user has access
    const deal = await prisma.deal.findUnique({
      where: { id },
      include: { stage: true },
    })
    if (!deal) return { status: 404, body: { error: 'Deal not found' } }
    if (user.role !== 'SALES_MANAGER' && deal.bdId !== user.id) {
      return { status: 403, body: { error: 'Not your deal' } }
    }

    // Validate target stage exists
    const targetStage = await prisma.pipelineStage.findUnique({ where: { id: stageId } })
    if (!targetStage) return { status: 400, body: { error: 'Invalid stage' } }

    // Block Closed Lost without remarks
    if (targetStage.name === 'Closed Lost' && !deal.remarks?.trim()) {
      return { status: 400, body: { error: 'Remarks required before closing as Lost' } }
    }

    // Transaction: update deal + close old audit + open new audit
    const updated = await prisma.$transaction(async (tx) => {
      // Close current audit log
      await tx.dealAuditLog.updateMany({
        where: { dealId: id, exitedAt: null },
        data: { exitedAt: new Date() },
      })

      // Open new audit log
      await tx.dealAuditLog.create({
        data: {
          dealId: id,
          stageId,
          enteredAt: new Date(),
          changedById: user.id,
          notes: notes ?? `Moved from ${deal.stage.name} to ${targetStage.name}`,
        },
      })

      // Update deal
      const isClosed = ['Closed Won', 'Closed Lost'].includes(targetStage.name)
      const result = await tx.deal.update({
        where: { id },
        data: {
          stageId,
          lastStageUpdateAt: new Date(),
          isClosed,
          closedDate: isClosed ? new Date() : undefined,
        },
        include: { stage: true, bd: true, client: true },
      })

      // Update projection probability
      const prob = STAGE_PROBABILITY[targetStage.name] ?? 0
      await tx.dealProjection.updateMany({
        where: { dealId: id },
        data: { probabilityPct: prob },
      })

      return result
    })

    // Emit event for downstream handlers
    await emit({
      topic: 'deal.stage.changed',
      data: {
        dealId: id,
        previousStageId: deal.stageId,
        previousStageName: deal.stage.name,
        newStageId: stageId,
        newStageName: targetStage.name,
        bdId: deal.bdId,
        changedById: user.id,
      },
    })

    logger.info(`Deal ${id} moved from ${deal.stage.name} to ${targetStage.name}`)
    return { status: 200, body: updated }
  } catch (error: any) {
    if (error?.name === 'AuthError') {
      return { status: 401, body: { error: error.message } }
    }
    logger.error('updateStage error', error)
    return { status: 500, body: { error: 'Internal server error' } }
  }
}

5.2.2 BD Dashboard Step (Day 2) — Key Metrics

// steps/api/dashboard/bd.step.ts
import { type Handlers, type StepConfig, logger } from 'motia'
import prisma from '../../../lib/db'
import { authenticate, AuthError } from '../../../lib/auth'

export const config: StepConfig = {
  type: 'api',
  name: 'BD Dashboard',
  description: 'BD member performance dashboard with 10 metrics',
  triggers: [{ type: 'api', method: 'GET', path: '/api/dashboard/bd' }],
}

export const handler: Handlers['api'] = async (req) => {
  try {
    const user = await authenticate(req.request)

    const quarter = Number(req.queryParams?.quarter) || Math.ceil((new Date().getMonth() + 1) / 3)
    const year = Number(req.queryParams?.year) || new Date().getFullYear()
    const bdId = (user.role === 'SALES_MANAGER' && req.queryParams?.bdId)
      ? req.queryParams.bdId : user.id

    const qStart = new Date(year, (quarter - 1) * 3, 1)
    const qEnd = new Date(year, quarter * 3, 0, 23, 59, 59)

    // 1. Deals Closed (won) this quarter
    const closedWonStage = await prisma.pipelineStage.findFirst({ where: { name: 'Closed Won' } })
    const dealsClosed = await prisma.deal.count({
      where: {
        bdId, stageId: closedWonStage?.id,
        closedDate: { gte: qStart, lte: qEnd },
      },
    })

    // 2. Deals Closed revenue
    const closedRevenue = await prisma.deal.aggregate({
      where: {
        bdId, stageId: closedWonStage?.id,
        closedDate: { gte: qStart, lte: qEnd },
      },
      _sum: { revenue: true },
    })

    // 3. Open Pipeline value
    const openPipeline = await prisma.deal.aggregate({
      where: { bdId, isClosed: false },
      _sum: { revenue: true },
      _count: true,
    })

    // 4. Quota attainment
    const target = await prisma.target.findFirst({
      where: { bdId, periodType: 'QUARTERLY' },
      include: { date: true },
    })
    const quota = target ? Number(target.quota) : 1750000
    const actual = Number(closedRevenue._sum.revenue ?? 0)
    const quotaAttainment = quota > 0 ? (actual / quota) * 100 : 0

    // 5. Sales Forecast (closed + weighted open pipeline)
    const projections = await prisma.dealProjection.findMany({
      where: { bdId, deal: { isClosed: false } },
    })
    const weightedPipeline = projections.reduce(
      (sum, p) => sum + Number(p.projectedAmount) * (Number(p.probabilityPct) / 100), 0
    )
    const salesForecast = actual + weightedPipeline

    // 6. Variance
    const salesVariance = quota - actual

    // 7. Monthly excess/deficit (simplified: actual vs monthly target)
    const currentMonth = new Date().getMonth()
    const monthlyQuota = quota / 3
    const monthsElapsed = currentMonth - (quarter - 1) * 3 + 1
    const expectedPace = monthlyQuota * monthsElapsed
    const monthlyExcessDeficit = actual - expectedPace

    // 8. Quarterly excess/deficit
    const quarterlyExcessDeficit = actual - quota

    // 9. Pipeline by stage
    const pipelineByStage = await prisma.$queryRaw`
      SELECT ps.name AS stage, COUNT(d.id)::int AS count,
             COALESCE(SUM(d.revenue), 0)::float AS value
      FROM deal d
      JOIN pipeline_stage ps ON d.stage_id = ps.id
      WHERE d.bd_id = ${bdId} AND d.is_closed = false
      GROUP BY ps.name ORDER BY ps.name
    `

    // 10. Open deals list
    const openDeals = await prisma.deal.findMany({
      where: { bdId, isClosed: false },
      include: { stage: true, client: true },
      orderBy: { lastStageUpdateAt: 'asc' },
      take: 20,
    })

    return {
      status: 200,
      body: {
        dealsClosed,
        closedRevenue: actual,
        openPipeline: {
          count: openPipeline._count,
          value: Number(openPipeline._sum.revenue ?? 0),
        },
        quotaAttainment: Math.round(quotaAttainment * 100) / 100,
        quota,
        salesForecast: Math.round(salesForecast),
        salesVariance,
        monthlyExcessDeficit: Math.round(monthlyExcessDeficit),
        quarterlyExcessDeficit: Math.round(quarterlyExcessDeficit),
        pipelineByStage,
        openDeals: openDeals.map(d => ({
          id: d.id,
          dealName: d.dealName,
          revenue: Number(d.revenue),
          stage: d.stage.name,
          client: d.client.name,
          lastStageUpdateAt: d.lastStageUpdateAt,
        })),
        quarter,
        year,
      },
    }
  } catch (error: any) {
    if (error?.name === 'AuthError') {
      return { status: 401, body: { error: error.message } }
    }
    logger.error('BD dashboard error', error)
    return { status: 500, body: { error: 'Internal server error' } }
  }
}
